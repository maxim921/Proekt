/****** Объект:  UserDefinedFunction [dbo].[fn_ExamGrade]    Дата создания скрипта: 05.11.2026 14:31:50 ******/ 
SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO
CREATE   FUNCTION [dbo].[fn_ExamGrade](@Score INT)
RETURNS NVARCHAR(30)
AS
BEGIN
    DECLARE @Result NVARCHAR(30);
    
    SET @Result = CASE 
        WHEN @Score IS NULL THEN N'Нет результата'
        WHEN @Score < 0 OR @Score > 100 THEN N'Ошибка данных'
        WHEN @Score BETWEEN 90 AND 100 THEN N'Отлично'
        WHEN @Score BETWEEN 75 AND 89 THEN N'Хорошо'
        WHEN @Score BETWEEN 60 AND 74 THEN N'Удовлетворительно'
        ELSE N'Неудовлетворительно'
    END;
    
    RETURN @Result;
END;
GO
